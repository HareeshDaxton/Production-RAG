# =============================================================================
# Production RAG — dependency manifest (convenience installer).
#   Install everything at once:   uv add -r requirements.txt
#
# NOTE: pyproject.toml + uv.lock stay the SOURCE OF TRUTH. The command above adds
# these as direct deps in pyproject and updates the lock. This file just lets you
# install in one shot instead of one-by-one.
#
# Covers Phases 0-8 (the full backend + the API the Next.js UI talks to).
# Phase 9 infra (service split + Postgres/pgvector) is listed COMMENTED at the
# bottom on purpose — installed when we reach it, not before.
#
# NOT installed by this file:
#   - Node packages for the UI    -> cd frontend && npm install
#   - Redis server                -> docker compose up -d redis
#   - Model weights (bge / cross-encoder / EasyOCR) -> downloaded on first use
# =============================================================================

# --- Web API / serving -------------------------------------------------------
fastapi
uvicorn[standard]
python-multipart       # multipart parsing for POST /v1/ingest/upload

# --- Config, models, logging -------------------------------------------------
pydantic
pydantic-settings
pyyaml
python-json-logger

# --- LLM / embeddings / structured output ------------------------------------
openai                 # final grounded generation (API)
instructor             # Pydantic-typed LLM output + partial streaming
tiktoken               # token counting / context budgeting
httpx                  # async HTTP (LLM + GitHub API)

# --- Retrieval: dense + sparse + rerank (Phase 2) ----------------------------
chromadb               # dense vector store
sentence-transformers  # local embeddings + cross-encoder reranker (pulls torch)
rank-bm25              # sparse keyword index

# --- Multi-format ingestion, 9 types (Phase 7) -------------------------------
python-frontmatter     # YAML front-matter in markdown
beautifulsoup4         # HTML parsing
lxml                   # fast HTML/XML backend; also backs the XML loader
markdown-it-py         # markdown structure / headings
pymupdf                # PDF text extraction + page rasterisation for OCR
python-docx            # DOCX paragraphs, heading styles, tables
pillow                 # image decoding for the OCR path
easyocr                # OCR for images + scanned PDF pages; reuses torch
# NOTE: csv / json loaders use the stdlib — no package required.

# --- Corpus fetching ---------------------------------------------------------
PyGithub               # GitHub issues fetching (scripts/fetch_corpus.py)

# --- Evaluation & auto-eval (Phases 4 & 6) -----------------------------------
numpy
pandas                 # eval tables / reports
scikit-learn           # metrics, similarity helpers

# --- Semantic cache (Phase 5) ------------------------------------------------
redis                  # client only; the server runs via docker compose

# --- Local judge (optional, Phase 3) -----------------------------------------
ollama                 # only needed if models.judge points at a local Ollama

# --- Dev tooling -------------------------------------------------------------
# Installed as a dev group, not with the line above:
#   uv add --dev pytest pytest-asyncio ruff

# --- Postgres / pgvector (Phase 9A) ------------------------------------------
# Only needed when stores.vector = pgvector or stores.relational = postgres; the
# default chroma+sqlite pair needs nothing. The pool extra is required, not optional:
# FastAPI's sync endpoints run in a threadpool and must not share one connection.
psycopg[binary,pool]   # server runs via docker compose (pgvector/pgvector:pg17)
# No `pgvector` package: embeddings are sent as '[...]' text and cast ::vector,
# which keeps the dependency surface smaller.
